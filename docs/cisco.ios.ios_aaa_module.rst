   - This module works with connection ``network_cli``. See https://docs.ansible.com/ansible/latest/network/user_guide/platform_ios.html



Examples
--------

.. code-block:: yaml

    # Using merged

    # Before state:
    # -------------
    #
    # router#show running-config | section ^aaa
    # aaa new-model
    # aaa session-id common

    - name: Add the method lists, a server group and a dynamic authorization client
      cisco.ios.ios_aaa:
        config:
          authentication:
            login:
              - name: ISE-LOGIN
                methods:
                  - group ISE-TAC
                  - local
            dot1x:
              - name: default
                methods:
                  - group ISE-RAD
          authorization:
            exec:
              - name: ISE-EXEC
                methods:
                  - group ISE-TAC
                  - local
                  - if-authenticated
            commands:
              - level: 15
                name: ISE-CMD
                methods:
                  - group ISE-TAC
                  - local
                  - if-authenticated
          accounting:
            exec:
              - name: ISE-ACC
                record: start-stop
                methods:
                  - group ISE-TAC
          group_servers:
            - type: radius
              name: ISE-RAD
              servers:
                - ISE-RAD-PRI
                - ISE-RAD-SEC
              source_interface: Loopback0
          dynamic_author:
            clients:
              - address: 192.0.2.10
                server_key:
                  key: Secret123
        state: merged

    # Task Output
    # -----------
    #
    # commands:
    #   - aaa authentication login ISE-LOGIN group ISE-TAC local
    #   - aaa authentication dot1x default group ISE-RAD
    #   - aaa authorization exec ISE-EXEC group ISE-TAC local if-authenticated
    #   - aaa authorization commands 15 ISE-CMD group ISE-TAC local if-authenticated
    #   - aaa accounting exec ISE-ACC start-stop group ISE-TAC
    #   - aaa group server radius ISE-RAD
    #   - server name ISE-RAD-PRI
    #   - server name ISE-RAD-SEC
    #   - ip radius source-interface loopback0
    #   - aaa server radius dynamic-author
    #   - client 192.0.2.10 server-key Secret123

    # After state:
    # ------------
    #
    # router#show running-config | section ^aaa
    # aaa new-model
    # aaa authentication login ISE-LOGIN group ISE-TAC local
    # aaa authentication dot1x default group ISE-RAD
    # aaa authorization exec ISE-EXEC group ISE-TAC local if-authenticated
    # aaa authorization commands 15 ISE-CMD group ISE-TAC local if-authenticated
    # aaa accounting exec ISE-ACC start-stop group ISE-TAC
    # aaa group server radius ISE-RAD
    #  server name ISE-RAD-PRI
    #  server name ISE-RAD-SEC
    #  ip radius source-interface Loopback0
    # aaa server radius dynamic-author
    #  client 192.0.2.10 server-key 6 <encrypted>
    # aaa session-id common

    # Using replaced

    # Before state:
    # -------------
    #
    # aaa group server radius ISE-RAD
    #  server name ISE-RAD-PRI
    #  server name ISE-RAD-SEC
    #  ip radius source-interface Loopback0

    - name: Replace the group server, the second server and the source interface go
      cisco.ios.ios_aaa:
        config:
          group_servers:
            - type: radius
              name: ISE-RAD
              servers:
                - ISE-RAD-PRI
        state: replaced

    # Task Output
    # -----------
    #
    # commands:
    #   - aaa group server radius ISE-RAD
    #   - no server name ISE-RAD-SEC
    #   - no ip radius source-interface

    # Using overridden

    # Before state:
    # -------------
    #
    # aaa authentication login ISE-LOGIN group ISE-TAC local
    # aaa authentication login LOCAL-ONLY local

    - name: Keep only the login list of the task
      cisco.ios.ios_aaa:
        config:
          authentication:
            login:
              - name: ISE-LOGIN
                methods:
                  - group ISE-TAC
                  - local
        state: overridden

    # Task Output
    # -----------
    #
    # commands:
    #   - no aaa authentication login LOCAL-ONLY

    # Using deleted

    - name: Remove a method list and a group server
      cisco.ios.ios_aaa:
        config:
          authentication:
            login:
              - name: LOCAL-ONLY
          group_servers:
            - type: radius
              name: ISE-RAD
        state: deleted

    # Task Output
    # -----------
    #
    # commands:
    #   - no aaa authentication login LOCAL-ONLY
    #   - no aaa group server radius ISE-RAD

    # Using gathered

    - name: Gather the AAA configuration
      cisco.ios.ios_aaa:
        state: gathered

    # Task Output
    # -----------
    #
    # gathered:
    #   new_model: true
    #   session_id: common
    #   authentication:
    #     login:
    #       - name: ISE-LOGIN
    #         methods:
    #           - group ISE-TAC
    #           - local
    #   group_servers:
    #     - type: radius
    #       name: ISE-RAD
    #       servers:
    #         - ISE-RAD-PRI

    # Using rendered

    - name: Render the AAA commands
      cisco.ios.ios_aaa:
        config:
          accounting:
            system:
              - name: default
                record: start-stop
                methods:
                  - group ISE-TAC
        state: rendered

    # Task Output
    # -----------
    #
    # rendered:
    #   - aaa accounting system default start-stop group ISE-TAC

    # Using parsed

    - name: Parse the AAA configuration
      cisco.ios.ios_aaa:
        running_config: |
          aaa new-model
          aaa authentication login default group ISE-TAC local
          aaa session-id common
        state: parsed

    # Task Output
    # -----------
    #
    # parsed:
    #   new_model: true
    #   session_id: common
    #   authentication:
    #     login:
    #       - name: default
    #         methods:
    #           - group ISE-TAC
    #           - local
