***************************
cisco.ios.ios_track_objects
***************************

**Resource module to configure tracked objects.**


Version added: 11.8.0

.. contents::
   :local:
   :depth: 1


Synopsis
--------
- This module configures and manages the attributes of tracked objects on Cisco IOS.

   - Tested against Cisco IOS XE Software, Version 17.12.4 on a Catalyst 9300.

Examples
--------

.. code-block:: yaml

    # Using merged

    # Before state:
    # -------------
    #
    # router#show running-config | section ^track
    # router#

    - name: Merge tracked objects
      cisco.ios.ios_track_objects:
        config:
          - object_id: 1
            ip_sla:
              operation_id: 1
              parameter: state
            delay:
              down: 120
              up: 180
          - object_id: 2
            ip_route:
              prefix: 192.0.2.0/28
              vrf: MGMT
            delay:
              down: 10
              up: 5
          - object_id: 10
            boolean_list:
              operator: and
              objects:
                - object_id: 1
                - object_id: 2
                  negate: true
            delay:
              down: 10
        state: merged

    # Task Output
    # -----------
    #
    # commands:
    #   - track 1 ip sla 1 state
    #   - delay down 120 up 180
    #   - track 2 ip route 192.0.2.0 255.255.255.240 reachability
    #   - ip vrf MGMT
    #   - delay down 10 up 5
    #   - track 10 list boolean and
    #   - object 1
    #   - object 2 not
    #   - delay down 10

    # After state:
    # ------------
    #
    # router#show running-config | section ^track
    # track 1 ip sla 1 state
    #  delay down 120 up 180
    # track 2 ip route 192.0.2.0 255.255.255.240 reachability
    #  ip vrf MGMT
    #  delay down 10 up 5
    # track 10 list boolean and
    #  object 1
    #  object 2 not
    #  delay down 10

    # Using replaced

    # Before state:
    # -------------
    #
    # track 1 ip sla 1 state
    #  delay down 120 up 180

    - name: Replace tracked object 1
      cisco.ios.ios_track_objects:
        config:
          - object_id: 1
            ip_sla:
              operation_id: 1
              parameter: state
            delay:
              up: 30
        state: replaced

    # Task Output
    # -----------
    #
    # commands:
    #   - track 1
    #   - no delay
    #   - delay up 30

    # Using overridden

    # Before state:
    # -------------
    #
    # track 1 ip sla 1 state
    # track 2 ip sla 2 state

    - name: Override tracked objects
      cisco.ios.ios_track_objects:
        config:
          - object_id: 1
            ip_sla:
              operation_id: 1
              parameter: state
        state: overridden

    # Task Output
    # -----------
    #
    # commands:
    #   - no track 2

    # Using deleted

    - name: Delete tracked object 1
      cisco.ios.ios_track_objects:
        config:
          - object_id: 1
        state: deleted

    # Task Output
    # -----------
    #
    # commands:
    #   - no track 1

    - name: Delete all tracked objects
      cisco.ios.ios_track_objects:
        state: deleted

    # Using gathered

    - name: Gather tracked objects
      cisco.ios.ios_track_objects:
        state: gathered

    # Task Output
    # -----------
    #
    # gathered:
    #   - object_id: 1
    #     ip_sla:
    #       operation_id: 1
    #       parameter: state
    #     delay:
    #       down: 120
    #       up: 180

    # Using rendered

    - name: Render tracked object commands
      cisco.ios.ios_track_objects:
        config:
          - object_id: 3
            interface:
              name: GigabitEthernet1
              parameter: line_protocol
        state: rendered

    # Task Output
    # -----------
    #
    # rendered:
    #   - track 3 interface GigabitEthernet1 line-protocol

    # Using parsed

    - name: Parse tracked objects
      cisco.ios.ios_track_objects:
        running_config: |
          track 1 ip sla 1 state
           delay down 120 up 180
        state: parsed

    # Task Output
    # -----------
    #
    # parsed:
    #   - object_id: 1
    #     ip_sla:
    #       operation_id: 1
    #       parameter: state
    #     delay:
    #       down: 120
    #       up: 180
